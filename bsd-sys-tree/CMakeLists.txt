
project(bsd-sys-tree LANGUAGES C)
if(NOT MSVC AND NOT (CMAKE_C_COMPILER_ID MATCHES "Watcom"))
    add_compile_definitions(_DEFAULT_SOURCE _XOPEN_SOURCE=700 _POSIX_C_SOURCE=200809L _DARWIN_C_SOURCE)
endif()

file(GLOB SOURCES "src/*.c")
if(SOURCES)
    add_library(bsd-sys-tree STATIC ${SOURCES})
    target_include_directories(bsd-sys-tree PUBLIC "include" "${CMAKE_CURRENT_SOURCE_DIR}/../bsd-sys-cdefs/include")
else()
    add_library(bsd-sys-tree INTERFACE)
    target_include_directories(bsd-sys-tree INTERFACE "include" "${CMAKE_CURRENT_SOURCE_DIR}/../bsd-sys-cdefs/include")
endif()

enable_testing()
file(GLOB TEST_SOURCES "tests/*.c")
add_executable(test_bsd-sys-tree ${TEST_SOURCES})
target_include_directories(test_bsd-sys-tree PRIVATE ${greatest_SOURCE_DIR})
target_link_libraries(test_bsd-sys-tree PRIVATE bsd-sys-tree)
add_test(NAME test_bsd-sys-tree COMMAND test_bsd-sys-tree)
